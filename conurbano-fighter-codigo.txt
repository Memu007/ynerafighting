      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
      if (slide !== 0) {

      }
      gain.gain.setValueAtTime(vol, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();

  function handleActionPress(code) {
    initAudio();

    if(gameState===STATE.CAREER_EVENT){
      if(['Digit1','KeyZ'].includes(code))chooseCareerEvent(0);

      return;
    }
    if (gameState === STATE.TITLE) {
      if(savedRun && code!=='KeyN'){
        resumeRun();return;

      gameState = STATE.INTRO_STORY;
      storyStep = 0;
      startBGM('madero');
      sfxSelect();

      storyStep++;

      if (storyStep >= 2) {
        currentFightIndex = 0;
        gameState = STATE.VS_SCREEN;
        vsTimer = 210;
        startBGM('vs');

    if (gameState === STATE.VS_SCREEN) {

      storyStep++;

    }
    if (gameState === STATE.ARRIVAL) {
      if (sceneFrame >= 240) { gameState = STATE.COACH; startBGM('gym'); }
      return;
    }
    if(gameState===STATE.BRIEF){gameState=STATE.GYM;saveProgress();sfxSelect();return;}
    if(gameState === STATE.TRAIN_TRAVEL){finishTravel();return;}
    if(gameState === STATE.WEEK_RESULT) {
      if(sceneFrame < 120) return;
      nextCareerWeek();return;
    }
    if (gameState === STATE.COACH) {
      playerStats.skillPoints = 0; playerStats.trainsLeft = 3;
      currentFightIndex = 1; week = 1; gameState = STATE.BRIEF;
      saveProgress();sfxSelect(); return;
    }
    if (gameState === STATE.GYM && minigame.active) {
      if(!['KeyZ','KeyX','Space'].includes(code))return;
      if(minigame.frames-minigame.lastInput<10)return;minigame.lastInput=minigame.frames;
      minigame.poseTimer=minigame.statTarget==='fuerza'?22:14;minigame.lastKey=code;
      if(minigame.statTarget==='tecnica'){
        const expected=minigame.sequence[minigame.attempts];
        if(code===expected || code==='Space'&&expected==='KeyZ'){minigame.hits++;sfxParry();}else sfxWeakHit();
        minigame.attempts++;if(minigame.attempts>=6)finishTraining();
      }else if(minigame.statTarget==='aguante'){
        const phase=minigame.frames%30;minigame.attempts++;
        if(phase>=8&&phase<=22){minigame.hits++;minigame.bagSwing=16;sfxHit();}else{minigame.bagSwing=6;sfxWeakHit();}
        if(minigame.hits>=12)finishTraining();
      }else if(minigame.statTarget==='velocidad'){
        if(!minigame.jumped){minigame.jumped=true;if(minigame.cursor>=38&&minigame.cursor<=62){minigame.hits++;sfxParry();}else sfxWeakHit();}
      }else{
        minigame.attempts++;
        if(minigame.cursor>=38&&minigame.cursor<=62){minigame.hits++;sfxParry();}else sfxWeakHit();
        minigame.speed+=.7;if(minigame.attempts>=3)finishTraining();
      }
      return;
    }

    if (gameState === STATE.VICTORY_FINAL) {
      resetRun();
      gameState = STATE.TITLE;
      sfxSelect();
      return;
    }

    // Acciones en combate
    if (gameState === STATE.FIGHT && player && (player.state === 'idle' || player.state === 'walk')) {
      if (code === 'KeyZ') {
        player.state = 'punch';
        player.stateTimer = Math.max(20, 26 - Math.min(6,playerStats.velocidad));
        executePlayerAttack('punch');
      } else if (code === 'KeyX') {
        player.state = 'kick';
        player.stateTimer = Math.max(28, 34 - Math.min(6,playerStats.velocidad));

      } else if (code === 'KeyV' && player.energy >= 100 && !FIGHTS[currentFightIndex].isPrologue) {
        player.energy = 0;
        player.state = 'special';
        player.stateTimer = 38;
        specialFx=38;specialOrigin=player.x;specialDirection=player.dir;
        executePlayerAttack('special');
      }
    }

  // 7. LÓGICA DE ATAQUES Y BALANCE DEL PRIMER RIVAL (KEVIN)
  // ========================================================
  function executePlayerAttack(type) {
    const f = FIGHTS[currentFightIndex];
    const dist = Math.abs(player.x - enemy.x);
    const reach = type === 'kick' ? 88 : (type === 'special' ? 115 : 70);

    if (dist <= reach && enemy.state !== 'ko') {
      // BALANCE DEL PRÓLOGO PEDIDO:
      // Podés pegarle 2, 3, 4 o 5 veces seguidas y entran los golpes, pero le sacan muy poquita vida (2 a 4 HP).
      // Después de que le conectás unos golpes intentando defenderte, Kevin se enoja y empieza a contratacar fuerte.
      if (f.isPrologue) {
        const chipDmg = type === 'kick' ? 4 : 3;
        enemy.hp = Math.max(180, enemy.hp - chipDmg);
        enemy.state = 'hit';
        enemy.stateTimer = 10; // Se sacude para que sientas que le pegaste
        enemy.vx = 2.2;addHitFx((player.x+enemy.x)/2,enemy.y-90,'light');
        prologueHitsLanded++;
        sfxWeakHit();
        addFloatingText(enemy.x, enemy.y - 85, `¡POCA FUERZA! -${chipDmg}`, '#ff9966');

        if (prologueHitsLanded === 3) {
          enemy.enraged = true;
          fightBanner = 'KEVIN: "¿ME QUERÉS PEGAR GIL? ¡AHORA VAS A VER!"';
          fightBannerTimer = 110;
        }
        return;
      }

      // Combates normales del Torneo y UFC
      let baseDmg = type === 'punch' ? 8 : (type === 'kick' ? 13 : 32);
      let totalDmg = Math.round(baseDmg + playerStats.fuerza * 1 + (type === 'special' ? playerStats.tecnica * 2 : 0));

      // Una secuencia repetida de golpes provoca guardia y contraataque.
      const style=f.style||'balanced';
      enemy.hitsTaken++;
      if(type !== 'special' && (enemy.state==='block' || (style==='counter'&&enemy.hitsTaken%4===0))) {
        totalDmg = Math.max(2, Math.round(totalDmg*(type==='kick'?.55:.22)));
        enemy.hp = Math.max(0,enemy.hp-totalDmg);addHitFx((player.x+enemy.x)/2,enemy.y-88,'block');
        enemy.state='block';enemy.stateTimer=8;enemy.telegraph=style==='counter'?30:18;
        enemy.aiCooldown=12;
        player.energy=Math.min(100,player.energy+18);
        addFloatingText(enemy.x,enemy.y-90,'¡GUARDIA!', '#66dd88');sfxBlock();return;
      }

      // Si el enemigo tenía cartel de alerta (!), metés ANTICIPO crítico
      if (enemy.telegraph > 0) {
        totalDmg = Math.round(totalDmg * (type==='special'?1.45:1.15));
        if(type==='special'){enemy.telegraph=0;enemy.aiCooldown=Math.max(enemy.aiCooldown,25);}
        addFloatingText(enemy.x, enemy.y - 95, `¡ANTICIPO CRÍTICO! -${totalDmg}`, '#00ffcc');
        sfxParry();
      } else {
        addFloatingText(enemy.x, enemy.y - 85, `-${totalDmg}`, '#ffcc00');
        sfxHit(type !== 'punch');
      }

      enemy.hp = Math.max(0, enemy.hp - totalDmg);addHitFx((player.x+enemy.x)/2,enemy.y-(type==='kick'?70:92),type==='special'?'special':type==='kick'||enemy.telegraph>0?'heavy':'light');
      if(enemy.telegraph===0){enemy.state='hit';enemy.stateTimer=type==='special'?20:7;}
      enemy.vx = player.dir * (type === 'kick' ? 6 : 3.5);
      if(type !== 'special') player.energy = Math.min(100, player.energy + 22);
      if(player.energy === 100 && type !== 'special'){fightBanner='¡ESPECIAL LISTA! APRETÁ V';fightBannerTimer=60;}
      player.combo++;
    } else if(type!=='special'&&!f.isPrologue){player.energy=Math.min(100,player.energy+4);}
  }

  function update() {
    if(document.hidden)return;
    if(gameState===STATE.TRAIN_TRAVEL){sceneFrame++;if(sceneFrame>=110)finishTravel();return;}
    if(gameState === STATE.WEEK_RESULT){sceneFrame++;return;}
    if (gameState === STATE.ARRIVAL) { sceneFrame++; if (sceneFrame >= 330) { gameState = STATE.COACH; startBGM('gym'); } return; }
    if (gameState === STATE.VS_SCREEN) {
      vsTimer--;
      if (vsTimer <= 0) {
        setupFight(currentFightIndex);
        gameState = STATE.FIGHT;
        focusArena();startBGM(FIGHTS[currentFightIndex].international ? 'ufc' : 'madero');
      }
      return;
    }

    if (gameState === STATE.GYM && minigame.active) {
      minigame.frames++;
      if(minigame.poseTimer>0)minigame.poseTimer--;
      minigame.bagSwing*=.94;
      if(minigame.statTarget==='aguante'){if(minigame.frames>=420)finishTraining();return;}
      if(minigame.statTarget==='tecnica')return;
      minigame.cursor+=minigame.dir*minigame.speed;
      if(minigame.statTarget==='velocidad'){
        if(minigame.cursor>=100){minigame.cursor=0;minigame.jumped=false;minigame.attempts++;if(minigame.attempts>=5)finishTraining();}
      }else if(minigame.cursor>=100){minigame.cursor=100;minigame.dir=-1;}else if(minigame.cursor<=0){minigame.cursor=0;minigame.dir=1;}
      return;
    }

    if (gameState !== STATE.FIGHT) return;

    const f = FIGHTS[currentFightIndex];
    hitFx.forEach(h=>h.life--);hitFx=hitFx.filter(h=>h.life>0);if(shake>0)shake--;
    if(player.state==='ko'||enemy.state==='ko')return;

    if(specialFx>0)specialFx--;
    // Reloj de combate
    lastSecondTick++;
    if (lastSecondTick >= 60 && fightTimer > 0) {
      fightTimer--;
      lastSecondTick = 0;
    }

    if(fightTimer===0 && player.hp>0 && enemy.hp>0){
      fightBanner='TIEMPO · DECISIÓN POR VIDA RESTANTE';fightBannerTimer=120;
      if(!f.isPrologue && player.hp/player.maxHp>enemy.hp/enemy.maxHp)enemy.hp=0;
      else player.hp=0;
    }
    if (fightBannerTimer > 0) fightBannerTimer--;

    // Textos flotantes
    floatingTexts.forEach(t => {
      t.y -= 0.8;
      t.life--;
    });
    floatingTexts = floatingTexts.filter(t => t.life > 0);

    // Actualizar Jugador
    if (player.stateTimer > 0) {
      player.stateTimer--;
      if (player.stateTimer === 0 && player.state !== 'ko') {
        player.state = 'idle';
      }
    }

    const moveSpeed = (2.6 + playerStats.velocidad * 0.45)*(1-(f.isPrologue?0:playerStats.fatigue/470));
    if (player.state === 'idle' || player.state === 'walk' || player.state === 'block') {
      if (keys['KeyC']) {
        player.state = 'block';
        player.vx = 0;
      } else if (keys['ArrowLeft']) {
        player.vx = -moveSpeed;
        player.state = 'walk';
      } else if (keys['ArrowRight']) {
        player.vx = moveSpeed;
        player.state = 'walk';
      } else {
        player.vx = 0;
        player.state = 'idle';
      }
    }

    player.x += player.vx;
    player.vx *= 0.8;
    player.x = Math.max(45, Math.min(595, player.x));
    player.dir = player.x <= enemy.x ? 1 : -1;

    // Actualizar Rival (IA)
    if (enemy.stateTimer > 0) {
      enemy.stateTimer--;
      if (enemy.stateTimer === 0 && enemy.state !== 'ko') {
        enemy.state = 'idle';
      }
    }

    if (enemy.telegraph > 0) {
      enemy.telegraph--;
      if (enemy.telegraph === 0 && enemy.state !== 'hit' && enemy.state !== 'ko') {
        // Lanza el golpe anunciado
        enemy.state = 'attack';
        enemy.stateTimer = 18;
        const dist = Math.abs(enemy.x - player.x);
        if (dist < (f.style==='longreach'?112:82)) {
          // En el prólogo: al principio te deja pegarle un par de veces, y cuando se enoja te saca de a 28-35 HP (te gana en ~3 o 4 golpes, no de una)
          let dmg = f.isPrologue ? (enemy.enraged ? 32 : 22) : f.atk;

          if (player.state === 'block') {
            const blockedDmg = f.isPrologue ? 12 : Math.max(2, Math.round(dmg * 0.22));
            player.hp = Math.max(0, player.hp - blockedDmg);addHitFx((player.x+enemy.x)/2,player.y-88,'block');
            player.energy = Math.min(100, player.energy + 15);
            sfxBlock();
            addFloatingText(player.x, player.y - 80, `GUARDIA -${blockedDmg}`, '#66dd88');
          } else {
            player.hp = Math.max(0, player.hp - dmg);addHitFx((player.x+enemy.x)/2,player.y-90,'heavy');
            player.state = 'hit';
            player.stateTimer = 16;
            player.combo = 0;
            if(!f.isPrologue) player.energy=Math.min(100,player.energy+12);
            sfxHit(true);
            addFloatingText(player.x, player.y - 85, `-${dmg}`, '#ff4444');
          }
        }
      }
    }

    // Each rival's announced style drives spacing, guard and attack timing.
    if(enemy.state==='idle' && enemy.telegraph===0 && player.hp>0 && enemy.hp>0){
      const dist=enemy.x-player.x, gap=Math.abs(dist), style=f.style||'balanced';
      enemy.dir=dist>=0?-1:1;
      if(f.isPrologue&&prologueHitsLanded>=2)enemy.enraged=true;
      const spacing=style==='longreach'?94:62;
      if(gap>spacing){enemy.x-=Math.sign(dist)*(f.isPrologue&&!enemy.enraged?1.5:f.speed);}
      else if(style==='longreach'&&gap<76){enemy.x+=Math.sign(dist)*f.speed*.65;}
      enemy.x=Math.max(45,Math.min(595,enemy.x));
      if(gap<=spacing+12){
        enemy.aiCooldown--;
        if((style==='defensive'||style==='balanced'&&enemy.attackCount%2===1)&&enemy.aiCooldown>20){enemy.state='block';enemy.stateTimer=14;enemy.aiCooldown-=8;}
        if(style==='counter'&&['punch','kick'].includes(player.state))enemy.aiCooldown=Math.min(enemy.aiCooldown,8);
        if(enemy.aiCooldown<=0){
          enemy.attackCount++;
          enemy.telegraph=f.isPrologue?18:style==='pressure'?14:style==='counter'?24:22;
          enemy.aiCooldown=style==='pressure'?(enemy.attackCount%3===0?75:32):style==='defensive'?55:style==='counter'?48:44;
        }
      }
    }

    // Emboscada de El Braian en el prólogo si ya le pegaste varias veces y te queda poca vida
    if (f.isPrologue && !braianAmbush.active && !braianAmbush.done && prologueHitsLanded >= 3 && player.hp <= 48) {
      braianAmbush.active = true;
      braianAmbush.x = Math.max(20, player.x - 140);
      fightBanner = '¡CUIDADO! ¡EL BRAIAN APARECE DE ATRÁS!';
      fightBannerTimer = 100;
    }

    if (braianAmbush && braianAmbush.active && !braianAmbush.done) {
      braianAmbush.x += 5.2;
      if (braianAmbush.x >= player.x - 25) {
        braianAmbush.done = true;
        player.hp = 0;addHitFx(player.x-10,player.y-100,'ko');
        sfxKO();
        addFloatingText(player.x, player.y - 95, '¡GOLPE TRAICIONERO DE ATRÁS!', '#ff2222');
      }
    }

    // Comprobar Fin del Combate
    if (player.hp <= 0 && player.state !== 'ko') {
      player.state = 'ko';addHitFx(player.x,player.y-90,'ko');
      sfxKO();
      setTimeout(() => {
        if (f.isPrologue) {
          gameState = STATE.POST_ROBO; storyStep = 0;
        } else {
          // Si perdés en el torneo, volvés al gimnasio con 1 entrenamiento extra de motivación
          recordResult(false,f.name);
        }
      }, 1700);
    } else if (enemy.hp <= 0 && enemy.state !== 'ko') {
      enemy.state = 'ko';addHitFx(enemy.x,enemy.y-90,'ko');
      sfxKO();
      fightBanner = '¡K.O. - GANADOR LAUTARO!';
      fightBannerTimer = 120;
      setTimeout(() => {
        if (f.isPrologue) {
          currentFightIndex = 1; gameState = STATE.POST_ROBO; storyStep = 0;
        } else {
          recordResult(true,f.name);
        }
      }, 1900);
    }
  }

  // Generador determinista para multitudes y detalles repetidos: el fondo no cambia entre cuadros.
  function seeded(seed){let s=seed>>>0||1;return()=>{s=(s*1664525+1013904223)>>>0;return s/4294967296;};}
  // `frame` alterna el balanceo de cada hincha sin cambiar la secuencia aleatoria de colores.
  function paintCrowd(g,x0,x1,y0,rows,seed,dark,frame=0){
    const rnd=seeded(seed);
    const skins=['#d8a57a','#b97d55','#8c5a3a','#e8bb93','#6f4330'],shirts=['#c23a3a','#2f4f9f','#e5e1d4','#2b2b33','#d8b63c','#3f7f4e','#7a3f8f','#d16a2a'];
    for(let r=0;r<rows;r++){
      const y=y0+r*12,fade=dark?.45+r*.18:1;
      for(let x=x0+(r%2)*7;x<x1;x+=14){
        const bob=(Math.floor(rnd()*3)+frame*(x/14%2?1:2))%3;
        g.globalAlpha=Math.min(1,fade);
        g.fillStyle=shirts[Math.floor(rnd()*shirts.length)];g.fillRect(x-5,y+7-bob,11,9);
        g.fillStyle=skins[Math.floor(rnd()*skins.length)];g.fillRect(x-3,y-bob,7,8);
        g.fillStyle='#1f1a1a';g.fillRect(x-3,y-bob-2,7,3);
      }
    }
    g.globalAlpha=1;
  }
  function paintSkyline(g,y,color,seed,lit){
    const rnd=seeded(seed);let x=-10;
    while(x<650){
      const w=24+Math.floor(rnd()*40),h=30+Math.floor(rnd()*80);

  function spritePose(f,phase){
    switch(f.state){
      case 'walk':return Math.sin(phase)>=0?'paso1':'paso2';
      case 'punch':case 'attack':return 'pina';
      case 'kick':return 'patada';
      case 'block':return 'bloqueo';
      case 'hit':return 'golpe';
      case 'ko':return 'ko';
      case 'special':return 'especial';
      case 'seated':return 'sentado';
      case 'point':return 'senala';
